> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Integrar la infraestructura Django↔n8n (`N8nClient`, `PendingJob`, callback endpoint), migrar `/estadisticas` y enriquecer `/prospecto` con workflows de n8n, y deprecar `/importcookies` y `/login`.

**Architecture:** Django despacha jobs a n8n vía `N8nClient` (HTTP POST al webhook), n8n ejecuta el workflow y devuelve el resultado con un POST a `/api/n8n/callback/` autenticado con token. El callback formatea el dato crudo con Gemini y lo envía por Telegram al usuario. El usuario recibe "⏳ Procesando..." de inmediato. `PendingJob` rastrea el estado de cada job. Los perfiles sociales en `/prospecto` (instagram.com, linkedin.com, facebook.com, tiktok.com) se despachan a n8n de forma async mientras los URLs no-sociales siguen con el scraping Playwright síncrono actual.

**Tech Stack:** Django 5.2, pytest, unittest.mock, django-rq, requests, n8n (ya corriendo en `prospector-n8n-1:5678`), Gemini (para formatear output de n8n), python-telegram-bot.

## Contrato n8n ↔ Django

**Django → n8n (dispatch):**
```json
{"job_id": "uuid-v4", "chat_id": "123456789", "params": {"url": "https://..."}}
```

**n8n → Django (callback):**
```json
{"job_id": "uuid-v4", "chat_id": "123456789", "status": "ok", "data": {...}}
```
Header: `X-N8N-Token: <token>`

# core/agent/infrastructure/n8n_client.py
import logging
import requests
from django.conf import settings

logger = logging.getLogger(__name__)


class N8nClient:
    def dispatch(self, workflow_id: str, params: dict, job_id: str, chat_id: str) -> None:
        """Envía un job a n8n via webhook. Lanza excepción si falla."""
        base_url = getattr(settings, 'N8N_BASE_URL', '')
        if not base_url:
            raise ValueError('N8N_BASE_URL no configurada en settings.')
        url = f'{base_url}/{workflow_id}'
        payload = {'job_id': job_id, 'chat_id': str(chat_id), 'params': params}
        resp = requests.post(url, json=payload, timeout=30)
        resp.raise_for_status()
```

# core/agent/interfaces/n8n_views.py
import json
import logging
from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.views.decorators.http import require_POST
from django.conf import settings
from django.utils import timezone

logger = logging.getLogger(__name__)


@csrf_exempt
@require_POST
def n8n_callback(request):
    token = request.headers.get('X-N8N-Token', '')
    expected = getattr(settings, 'N8N_CALLBACK_TOKEN', '')
    if not expected or token != expected:
        return JsonResponse({'error': 'Unauthorized'}, status=401)

    try:
        body = json.loads(request.body)
    except json.JSONDecodeError:
        return JsonResponse({'error': 'Invalid JSON'}, status=400)

    job_id = body.get('job_id')
    chat_id = body.get('chat_id')
    status = body.get('status', 'ok')
    data = body.get('data', {})

    if not job_id or not chat_id:
        return JsonResponse({'error': 'Missing job_id or chat_id'}, status=400)

    from core.agent.infrastructure.models import PendingJob
    try:
        job = PendingJob.objects.get(job_id=job_id)
    except PendingJob.DoesNotExist:
        return JsonResponse({'error': 'Job not found'}, status=404)

    if status != 'ok':
        job.status = 'failed'
        job.completed_at = timezone.now()
        job.save()
        _send_telegram(chat_id, f'❌ Error en la operación: {data.get("error", "error desconocido")}')
        return JsonResponse({'ok': True})

    job.status = 'completed'
    job.completed_at = timezone.now()
    job.save()

    formatted = _format_with_gemini(job.command, data)
    _send_telegram(chat_id, formatted)
    return JsonResponse({'ok': True})


def _format_with_gemini(command: str, data: dict) -> str:
    from core.agent.infrastructure.gemini_adapter import GeminiAdapter
    gemini = GeminiAdapter()
    api_key = settings.GEMINI_API_KEY
    model = getattr(settings, 'AI_MODEL', 'gemini-2.5-flash')
    prompt = (
        f'Eres el asistente de negocio del usuario. '
        f'Formatea el siguiente resultado del comando /{command} en español, '
        f'de forma clara y concisa para Telegram (usa Markdown, máximo 3000 caracteres):\n\n'
        f'{json.dumps(data, ensure_ascii=False, indent=2)}'
    )
    try:
        return gemini.generate_response(prompt=prompt, api_key=api_key, model_name=model)
    except Exception as e:
        logger.error(f'Error formateando con Gemini en callback: {e}')
        return f'✅ Resultado recibido:\n```\n{json.dumps(data, ensure_ascii=False, indent=2)[:2000]}\n```'


def _send_telegram(chat_id: str, text: str) -> None:
    import requests
    token = settings.TELEGRAM_BOT_TOKEN
    if not token:
        return
    try:
        requests.post(
            f'https://api.telegram.org/bot{token}/sendMessage',
            json={'chat_id': chat_id, 'text': text, 'parse_mode': 'Markdown'},
            timeout=10,
        )
    except Exception as e:
        logger.error(f'Error enviando Telegram desde callback: {e}')
```

El usuario **solo necesita la UI** para: añadir credenciales (Instagram Token, LinkedIn OAuth2, etc.) y conectarlas al nodo Code de cada workflow. El resto ya viene configurado.

#!/usr/bin/env python3
"""
Despliega los 8 workflows de Sprint 12 en n8n via REST API.
Requiere: N8N_API_KEY y N8N_CALLBACK_TOKEN en el entorno o .env
Uso: python n8n_workflows/deploy.py
"""
import json
import os
import sys
import uuid
import requests

# ── Configuración ─────────────────────────────────────────────────────────────

def _load_env():
    env_path = os.path.join(os.path.dirname(__file__), '..', '.env')
    env = {}
    try:
        with open(env_path) as f:
            for line in f:
                line = line.strip()
                if line and not line.startswith('#') and '=' in line:
                    k, _, v = line.partition('=')
                    env[k.strip()] = v.strip()
    except FileNotFoundError:
        pass
    return env

_env = _load_env()
N8N_API_KEY = os.environ.get('N8N_API_KEY') or _env.get('N8N_API_KEY', '')
N8N_CALLBACK_TOKEN = os.environ.get('N8N_CALLBACK_TOKEN') or _env.get('N8N_CALLBACK_TOKEN', '')
N8N_BASE = 'http://localhost:5678'
DJANGO_CALLBACK_URL = 'http://172.17.0.1:3001/api/v1/agent/n8n/callback/'

if not N8N_API_KEY:
    print('ERROR: N8N_API_KEY no encontrada en .env ni en el entorno.')
    sys.exit(1)
if not N8N_CALLBACK_TOKEN:
    print('ERROR: N8N_CALLBACK_TOKEN no encontrada. Genera una con:')
    print('  python3 -c "import secrets; print(secrets.token_hex(32))"')
    sys.exit(1)

HEADERS = {'X-N8N-API-KEY': N8N_API_KEY, 'Content-Type': 'application/json'}

# ── Constructores de nodos ────────────────────────────────────────────────────

def webhook_node(workflow_id: str, pos_x=240, pos_y=300):
    wh_id = str(uuid.uuid4())
    return {
        'id': str(uuid.uuid4()),
        'name': 'Webhook',
        'type': 'n8n-nodes-base.webhook',
        'typeVersion': 2,
        'position': [pos_x, pos_y],
        'webhookId': wh_id,
        'parameters': {
            'httpMethod': 'POST',
            'path': workflow_id,
            'responseMode': 'onReceived',
            'options': {},
        },
    }


def social_placeholder_node(platform: str, workflow_type: str, pos_x=500, pos_y=300):
    """
    Nodo Code con placeholder. El usuario reemplaza el contenido con la
    llamada real a la API social una vez configuradas las credenciales en n8n.
    """
    if workflow_type == 'stats':
        data_shape = '''{
    platform: "PLATFORM",
    url: params.url,
    likes: null,
    comments: null,
    views: null,
    note: "Reemplaza este nodo con la llamada real a la API de PLATFORM"
  }'''.replace('PLATFORM', platform)
    else:
        data_shape = '''{
    platform: "PLATFORM",
    url: params.url,
    username: null,
    followers: null,
    bio: null,
    recent_posts: [],
    note: "Reemplaza este nodo con la llamada real a la API de PLATFORM"
  }'''.replace('PLATFORM', platform)

    code = f"""// Placeholder {platform}_{workflow_type}
// Añade tus credenciales de {platform.capitalize()} en n8n Settings > Credentials
// y reemplaza este código con la llamada real a la API.
const {{ job_id, chat_id, params }} = $input.first().json;
return [{{ json: {{ job_id, chat_id, data: {data_shape} }} }}];
"""
    return {
        'id': str(uuid.uuid4()),
        'name': f'API {platform.capitalize()}',
        'type': 'n8n-nodes-base.code',
        'typeVersion': 2,
        'position': [pos_x, pos_y],
        'parameters': {'jsCode': code, 'mode': 'runOnceForAllItems'},
    }


def callback_node(pos_x=760, pos_y=300):
    code = f"""// Envía resultado a Django via callback
const {{ job_id, chat_id, data }} = $input.first().json;
const payload = {{ job_id, chat_id, status: 'ok', data }};
try {{
  const resp = await fetch('{DJANGO_CALLBACK_URL}', {{
    method: 'POST',
    headers: {{
      'Content-Type': 'application/json',
      'X-N8N-Token': '{N8N_CALLBACK_TOKEN}',
    }},
    body: JSON.stringify(payload),
  }});
  return [{{ json: {{ callback_status: resp.status }} }}];
}} catch (err) {{
  return [{{ json: {{ callback_error: err.message }} }}];
}}
"""
    return {
        'id': str(uuid.uuid4()),
        'name': 'Django Callback',
        'type': 'n8n-nodes-base.code',
        'typeVersion': 2,
        'position': [pos_x, pos_y],
        'parameters': {'jsCode': code, 'mode': 'runOnceForAllItems'},
    }


def build_workflow(workflow_id: str, platform: str, workflow_type: str) -> dict:
    wh = webhook_node(workflow_id)
    social = social_placeholder_node(platform, workflow_type)
    cb = callback_node()
    return {
        'name': workflow_id,
        'nodes': [wh, social, cb],
        'connections': {
            'Webhook': {'main': [[{'node': social['name'], 'type': 'main', 'index': 0}]]},
            social['name']: {'main': [[{'node': 'Django Callback', 'type': 'main', 'index': 0}]]},
        },
        'active': False,
        'settings': {'executionOrder': 'v1'},
    }


# ── Workflows a crear ─────────────────────────────────────────────────────────

WORKFLOWS = [
    ('instagram_stats',      'instagram', 'stats'),
    ('tiktok_stats',         'tiktok',    'stats'),
    ('linkedin_stats',       'linkedin',  'stats'),
    ('facebook_stats',       'facebook',  'stats'),
    ('instagram_competitor', 'instagram', 'competitor'),
    ('linkedin_competitor',  'linkedin',  'competitor'),
    ('facebook_competitor',  'facebook',  'competitor'),
    ('tiktok_competitor',    'tiktok',    'competitor'),
]

# ── Deploy ────────────────────────────────────────────────────────────────────

def get_existing_workflow_names() -> set:
    resp = requests.get(f'{N8N_BASE}/api/v1/workflows', headers=HEADERS, timeout=10)
    resp.raise_for_status()
    return {w['name'] for w in resp.json().get('data', [])}


def create_workflow(wf_dict: dict) -> str:
    resp = requests.post(
        f'{N8N_BASE}/api/v1/workflows',
        headers=HEADERS,
        json=wf_dict,
        timeout=15,
    )
    resp.raise_for_status()
    return resp.json()['id']


def activate_workflow(wf_id: str) -> None:
    resp = requests.post(
        f'{N8N_BASE}/api/v1/workflows/{wf_id}/activate',
        headers=HEADERS,
        timeout=10,
    )
    resp.raise_for_status()


def main():
    print(f'Conectando a n8n en {N8N_BASE}...')
    existing = get_existing_workflow_names()
    print(f'Workflows existentes: {len(existing)}')

    created = []
    skipped = []
    for workflow_id, platform, wf_type in WORKFLOWS:
        if workflow_id in existing:
            skipped.append(workflow_id)
            print(f'  ⏭  {workflow_id} ya existe, omitiendo')
            continue
        wf_dict = build_workflow(workflow_id, platform, wf_type)
        wf_id = create_workflow(wf_dict)
        activate_workflow(wf_id)
        created.append(workflow_id)
        print(f'  ✅ {workflow_id} creado y activado (id={wf_id})')

    print(f'\nResumen: {len(created)} creados, {len(skipped)} ya existían.')
    print(f'Webhook base URL: http://localhost:5678/webhook/<workflow_id>')
    print('\nPróximo paso: configura credenciales de cada red social en')
    print(f'  {N8N_BASE}/credentials')
    print('y reemplaza el nodo "API <Red>" con la llamada real.')


if __name__ == '__main__':
    main()
